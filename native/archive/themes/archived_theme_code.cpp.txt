// ===== src/ui/look.cpp: Synthwave's purple glow behind a window's top =====
void drawSynthGlow(ImDrawList* dl, ImVec2 a, ImVec2 b) {
    // radial-gradient(130% 55% at 50% 0%, #3b1466 0%, var(--bg) 62%): soft rings from the top middle
    const app::Rgb glow = g_look.c.style.glowTop.c;
    const float w = b.x - a.x, h = b.y - a.y, rx = 1.3f * w * 0.62f, ry = 0.55f * h * 0.62f;
    for (int i = 24; i >= 1; i--) {
        const float t = i / 24.f;
        const ImU32 c = col(glow, 0.09f * (1 - t) + 0.02f);
        dl->PathClear();
        for (int k = 0; k <= 32; k++) {
            const float an = 3.14159265f * k / 32;
            dl->PathLineTo(ImVec2(a.x + w / 2 + rx * t * std::cos(an), a.y + ry * t * std::sin(an)));
        }
        dl->PathFillConvex(c);
    }
}

// ===== src/app/theme_style.cpp: withoutEffects() (Theme effects Off) =====
ThemeStyle withoutEffects(const ThemeStyle& s) {
    ThemeStyle o = s;
    PageFx& f = o.page.fx;
    f.clouds = f.bubbles = f.grain = f.halftone = f.grid = f.streak = 0;
    o.widget.glint = 0; o.widget.brush = 0;
    o.panel.reflectAlpha = 0; o.panel.reflectHeight = 0;
    o.panel.bracketAlpha = 0; o.panel.bracketSize = 0; o.panel.hudAlpha = 0; o.panel.spec.clear();
    return o;
}
